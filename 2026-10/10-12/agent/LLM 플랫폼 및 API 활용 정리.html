6. GPT, Gemini, Claude 중 선택해서 실행

from openai import OpenAI
from google import genai
from anthropic import Anthropic

def ask_ai(platform: str, key: str, question: str) -> str:
    if platform == "gpt":
        client = OpenAI(api_key=key)
        return client.responses.create(
            model="gpt-5.4", input=question
        ).output_text
    elif platform == "gemini":
        client = genai.Client(api_key=key)
        return client.models.generate_content(
            model="gemini-2.5-flash", contents=question
        ).text
    elif platform == "claude":
        client = Anthropic(api_key=key)
        result = client.messages.create(
            model="claude-sonnet-4-5", max_tokens=300,
            messages=[{"role": "user", "content": question}]
        )
        return result.content[0].text
    raise ValueError("지원하지 않는 플랫폼")

platform = input("플랫폼(gpt/gemini/claude): ")
key = input("API 키: ")
question = input("질문: ")
print(ask_ai(platform, key, question))

8. 플랫폼과 모델명 직접 선택

아래 코드는 7번에서 만든 각 클라이언트 클래스가 (model, api_key) 생성자와 ask() 메서드를 구현했다는 전제의 설계 예제입니다.

# main.py
from llm.factory import create_client

platform = input("플랫폼: ").strip().lower()
model = input("모델명: ").strip()
api_key = input("API 키: ").strip()
question = input("질문: ")
client = create_client(platform, model, api_key)
print(client.ask(question))
# llm/factory.py
from .openai_client import GPTClient
from .gemini_client import GeminiClient
from .claude_client import ClaudeClient

def create_client(platform: str, model: str, api_key: str):
    clients = {
        "gpt": GPTClient,
        "gemini": GeminiClient,
        "claude": ClaudeClient,
    }
    if platform not in clients:
        raise ValueError("지원하지 않는 플랫폼")
    return clients[platform](model=model, api_key=api_key)

실제 운영에서는 API 키를 .env에 저장하고 .gitignore로 제외하세요.

12. LangChain 체인 기본 예제

사용자 질문
↓
프롬프트 템플릿
↓
LLM
↓
문자열 출력
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser

llm = ChatOpenAI(model="gpt-5.4")
prompt = ChatPromptTemplate.from_template(
    "{topic}을 초보자에게 설명해줘"
)
chain = prompt | llm | StrOutputParser()
result = chain.invoke({"topic": "Docker"})
print(result)

|는 앞 단계의 출력을 다음 단계로 전달하는 연결 연산자입니다.

13. LangChain 체인 확장

주제 입력
↓
체인 1: 상세 설명
↓
체인 2: 3줄 요약
↓
최종 결과
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser

llm = ChatOpenAI(model="gpt-5.4")
parser = StrOutputParser()
explain = (
    ChatPromptTemplate.from_template("{topic}을 자세히 설명해줘")
    | llm | parser
)
summary = (
    ChatPromptTemplate.from_template("다음 내용을 3줄로 요약해줘:\n{text}")
    | llm | parser
)
result = explain.invoke({"topic": "Kubernetes"})
final = summary.invoke({"text": result})
print(final)

체인이 길어지면 API 호출 횟수, 비용, 지연 시간이 증가할 수 있습니다.

LangChain은 필수가 아니며 프로젝트 복잡도에 맞게 선택하는 것이 좋습니다.

기능설명
